# Планер дня — Android

Обёртка над веб-планером. Внутри WebView открывается страница из app/src/main/assets/index.html.
Будильники и напоминания ставятся системой через мост AndroidPlaner.

Сборка: при каждом коммите GitHub Actions собирает APK и публикует его в Releases.

const KEY = 'planer.v1';
const ACC = {purple:'#8b5cf6', blue:'#3b82f6', emerald:'#10b981', pink:'#ec4899', amber:'#f59e0b', teal:'#14b8a6'};

const DAYS = [['Пн',1],['Вт',2],['Ср',3],['Чт',4],['Пт',5],['Сб',6],['Вс',0]];
const DEF = {tasks:[], events:[], alarms:[], moods:{}, sleeps:[], accent:'purple', filter:'all', sleepNow:null, hiddenAt:null, selDay:null};

let S = Object.assign({}, DEF);
let docRef = null;
try { S = Object.assign({}, DEF, JSON.parse(localStorage.getItem(KEY)) || {}); } catch(e) {}
/* Мост к Android: если страница открыта в приложении, будильники и напоминания ставит система */
function syncNative(){
  if (!window.AndroidPlaner) return;
  try { AndroidPlaner.sync(JSON.stringify({alarms:S.alarms, events:S.events})); } catch(e) {}
}
let writeChain = Promise.resolve();
const save = () => {
  localStorage.setItem(KEY, JSON.stringify(S));
  syncNative();
  if (docRef) {
    const data = JSON.parse(JSON.stringify(S));
    writeChain = writeChain.then(() => docRef.set(data)).catch(() => {});

function bindDrag(){
  document.querySelectorAll('#tl .ev').forEach(el => {
    el.oncontextmenu = e => e.preventDefault();
    el.onpointerdown = e => {
      drag = {el, id: el.dataset.id, y: e.clientY, start: +el.dataset.start, dur: +el.dataset.dur, moved: false, cur: null, timer: null};
      el.setPointerCapture(e.pointerId); el.classList.add('drag');
      drag.timer = setTimeout(() => {
        if (!drag || drag.el !== el || drag.moved) return;
        const id = drag.id;
        endDrag();
        openEventMenu(id);
      }, 500);
    };
    el.onpointermove = e => {
      if (!drag || drag.el !== el) return;
      const dy = e.clientY - drag.y;
      if (Math.abs(dy) > 4 && !drag.moved) { drag.moved = true; clearTimeout(drag.timer); }
      let s = Math.round((drag.start + dy)/15)*15;               // шаг 15 минут, 1px = 1 минута
      s = Math.max(360, Math.min(1380 - drag.dur, s));
      drag.cur = s; el.style.top = (s-360) + 'px';
    };
    el.onpointerup = () => {
      if (!drag) return;
      clearTimeout(drag.timer);
      if (drag.moved && drag.cur != null){
        const ev = S.events.find(x => x.id === drag.id);
        if (ev){ ev.start = drag.cur; save(); }
      }
      drag = null; renderDay();
    };
  });
}

function addSleep(start, end, auto){
  const minutes = (end-start)/60000;
  if (minutes < 1) return;
  S.sleeps.push({id:uid(), date:dayKey(new Date(end)), start, end, minutes, auto});
  save();

function renderSleep(){
  const s = S.sleepNow;
  let html;
  if (!s){
    html = `<div class="h">Трекер сна</div>
      <div class="big" style="font-size:30px">Не спишь</div>
      <div class="muted" style="margin:6px 0 14px">Нажми, когда ложишься</div>

  } else if (s.phase === 'falling'){

    html = `<div class="h">Засыпаю</div>
      <div class="big" style="font-size:30px">${Math.ceil(left/60000)} мин</div>

function beep(){
  actx = actx || new (window.AudioContext || window.webkitAudioContext)();
  let n = 0;
  const iv = setInterval(() => {
    if (actx.state === 'suspended') actx.resume();
    const o = actx.createOscillator(), g = actx.createGain();
    o.type = 'sine'; o.frequency.value = 880;
    g.gain.setValueAtTime(.25, actx.currentTime);
    g.gain.exponentialRampToValueAtTime(.001, actx.currentTime + .35);
    o.connect(g).connect(actx.destination);
    o.start(); o.stop(actx.currentTime + .4);
    if (++n >= 40) clearInterval(iv);
  }, 700);
  return () => clearInterval(iv);
}

function fireAlarm(a){
  const t = $('#toast');
  t.innerHTML = `<div class="h">Будильник</div>
    <div class="row" style="justify-content:space-between"><span style="font-size:18px">${esc(a.label || 'Подъём')}</span>
    <button class="btn" id="stopAl">Стоп</button></div>`;
  t.style.display = 'block';
  stopBeep = beep();
  $('#stopAl').onclick = () => { stopBeep && stopBeep(); t.style.display = 'none'; };
}

function checkAlarms(){
  if (window.AndroidPlaner) return; // в приложении звонит нативный будильник
  const n = new Date();
  S.alarms.forEach(a => {
    if (!a.on || a.h !== n.getHours() || a.m !== n.getMinutes()) return;
    if (a.days.length && !a.days.includes(n.getDay())) return;
    const k = a.id + dayKey(n) + a.h + a.m;
    if (lastFire === k) return;
    lastFire = k;
    if (!a.days.length){ a.on = false; save(); }
    fireAlarm(a);
    renderAlarms();
  });
}

function tickSleep(){
  const s = S.sleepNow;
  if (s && s.phase === 'falling' && Date.now() - s.fallStart >= FALL_MS){
    s.phase = 'asleep'; s.sleepStart = s.fallStart + FALL_MS; save();
  }
}

    S.hiddenAt = null; save();
    renderSleep();
  }
});

package com.planer.day

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.net.Uri
import org.json.JSONArray
import org.json.JSONObject
import java.time.LocalDate
import java.time.LocalDateTime
import java.time.ZoneId

/**
 * Хранит будильники и напоминания, присланные страницей, и ставит их в системе.
 * Будильники — через setAlarmClock: работают при погашенном экране и в режиме Doze,
 * отдельного разрешения не требуют.
 */
object AlarmStore {
    private const val PREFS = "planer_native"
    private const val K_ALARMS = "alarms"
    private const val K_EVENTS = "events"
    private const val K_IDS = "scheduled"
    private const val K_FIRED = "fired"

    private fun prefs(ctx: Context) = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
    private fun am(ctx: Context) = ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager

    /** Полностью пересоздаёт все будильники и напоминания по данным страницы. */
    fun sync(ctx: Context, json: String): String {
        val root = JSONObject(json)
        val alarms = root.optJSONArray("alarms") ?: JSONArray()
        val events = root.optJSONArray("events") ?: JSONArray()

        cancelAll(ctx)
        prefs(ctx).edit()
            .putString(K_ALARMS, alarms.toString())
            .putString(K_EVENTS, events.toString())
            .apply()

        val keys = mutableSetOf<String>()
        for (i in 0 until alarms.length()) {
            val a = alarms.getJSONObject(i)
            if (!a.optBoolean("on", true)) continue
            if (scheduleAlarm(ctx, a)) keys.add("a" + a.getString("id"))
        }
        for (i in 0 until events.length()) {
            val e = events.getJSONObject(i)
            val at = eventTime(e) ?: continue
            if (at <= System.currentTimeMillis()) continue
            scheduleReminder(ctx, e.getString("id"), e.getString("title"), at)
            keys.add("e" + e.getString("id"))
        }
        prefs(ctx).edit().putStringSet(K_IDS, keys).apply()
        return "будильников ${alarms.length()}, напоминаний ${events.length()}"
    }

    /** Вызывается после перезагрузки телефона: ставим всё заново из сохранённого. */
    fun rescheduleStored(ctx: Context) {
        val p = prefs(ctx)
        val json = JSONObject()
            .put("alarms", JSONArray(p.getString(K_ALARMS, "[]")))
            .put("events", JSONArray(p.getString(K_EVENTS, "[]")))
        sync(ctx, json.toString())
    }

    /** После срабатывания: повторяющийся будильник ставим на следующий раз, разовый помечаем. */
    fun afterFire(ctx: Context, key: String) {
        val id = key.removePrefix("a")
        val arr = JSONArray(prefs(ctx).getString(K_ALARMS, "[]"))
        for (i in 0 until arr.length()) {
            val a = arr.getJSONObject(i)
            if (a.getString("id") != id) continue
            val days = a.optJSONArray("days")
            if (days == null || days.length() == 0) {
                val fired = (prefs(ctx).getStringSet(K_FIRED, emptySet()) ?: emptySet()).toMutableSet()
                fired.add(id)
                prefs(ctx).edit().putStringSet(K_FIRED, fired).apply()
            } else {
                scheduleAlarm(ctx, a)
            }
            return
        }
    }

    /** Страница забирает список сработавших разовых будильников, чтобы выключить их у себя. */
    fun takeFired(ctx: Context): String {
        val set = prefs(ctx).getStringSet(K_FIRED, emptySet()) ?: emptySet()
        prefs(ctx).edit().remove(K_FIRED).apply()
        return JSONArray(set.toList()).toString()
    }

    private fun cancelAll(ctx: Context) {
        val keys = prefs(ctx).getStringSet(K_IDS, emptySet()) ?: emptySet()
        for (k in keys) {
            val pi = buildPi(ctx, k, null, null, PendingIntent.FLAG_NO_CREATE) ?: continue
            am(ctx).cancel(pi)
            pi.cancel()
        }
    }

    private fun buildPi(ctx: Context, key: String, title: String?, label: String?, flags: Int): PendingIntent? {
        val cls = if (key.startsWith("a")) AlarmReceiver::class.java else ReminderReceiver::class.java
        val intent = Intent(ctx, cls)
            .setData(Uri.parse("planer://$key"))
            .putExtra("key", key)
        title?.let { intent.putExtra("title", it) }
        label?.let { intent.putExtra("label", it) }
        return PendingIntent.getBroadcast(ctx, key.hashCode(), intent, flags or PendingIntent.FLAG_IMMUTABLE)
    }

    private fun scheduleAlarm(ctx: Context, a: JSONObject): Boolean {
        val at = nextTime(a.getInt("h"), a.getInt("m"), a.optJSONArray("days")) ?: return false
        val key = "a" + a.getString("id")
        val pi = buildPi(ctx, key, null, a.optString("label", ""), PendingIntent.FLAG_UPDATE_CURRENT) ?: return false
        val showIntent = PendingIntent.getActivity(
            ctx, 0, Intent(ctx, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE
        )
        try {
            am(ctx).setAlarmClock(AlarmManager.AlarmClockInfo(at, showIntent), pi)
        } catch (e: SecurityException) {
            am(ctx).setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, pi)
        }
        return true
    }

    private fun scheduleReminder(ctx: Context, id: String, title: String, at: Long) {
        val pi = buildPi(ctx, "e$id", title, null, PendingIntent.FLAG_UPDATE_CURRENT) ?: return
        am(ctx).setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, pi)
    }

    /** Ближайшее будущее время h:m с учётом дней недели (0 = воскресенье, как в JS). */
    private fun nextTime(h: Int, m: Int, days: JSONArray?): Long? {
        val now = LocalDateTime.now()
        for (d in 0..7) {
            val cand = LocalDate.now().plusDays(d.toLong()).atTime(h, m)
            if (!cand.isAfter(now)) continue
            val pageDay = cand.dayOfWeek.value % 7
            if (days != null && days.length() > 0) {
                var ok = false
                for (i in 0 until days.length()) {
                    if (days.getInt(i) == pageDay) ok = true
                }
                if (!ok) continue
            }
            return cand.atZone(ZoneId.systemDefault()).toInstant().toEpochMilli()
        }
        return null
    }

    private fun eventTime(e: JSONObject): Long? = try {
        LocalDate.parse(e.getString("date"))
            .atStartOfDay()
            .plusMinutes(e.getInt("start").toLong())
            .atZone(ZoneId.systemDefault())
            .toInstant()
            .toEpochMilli()
    } catch (ex: Exception) {
        null
    }
}

            textSize = size

            gravity = Gravity.CENTER

        stepView = text(14f, "#99ffffff", dp(8))
        taskView = text(24f, "#ffffff", dp(24))

        input = EditText(this).apply {
            setTextColor(Color.WHITE)
            setHintTextColor(Color.parseColor("#66ffffff"))
            hint = "Ответ"
            textSize = 20f
            gravity = Gravity.CENTER
            setSingleLine(true)
            setOnEditorActionListener { _, _, _ ->
                check()
                true
            }
        }

        val btn = Button(this).apply {
            text = "Проверить"
            setOnClickListener { check() }
        }

        root.addView(title)
        root.addView(stepView)
        root.addView(taskView)
        root.addView(input)
        root.addView(btn)

        input.setText("")

            val a = Random.nextInt(100, 1000)

            taskView.text = "Сколько будет $a + $b ?"
            input.inputType = InputType.TYPE_CLASS_NUMBER
        } else {
            expected = PHRASES.random()
            taskView.text = "Напечатай:\n$expected"

            NotificationChannel(CH_REMIND, "Напоминания", NotificationManager.IMPORTANCE_DEFAULT)
        )
    }

            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        val pi = PendingIntent.getActivity(
            ctx, notifId(key), full,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val builder = NotificationCompat.Builder(ctx, CH_ALARM)
            .setSmallIcon(android.R.drawable.ic_lock_idle_alarm)
            .setContentTitle(label.ifBlank { "Будильник" })
            .setContentText("Реши задачку, чтобы выключить")
            .setPriority(NotificationCompat.PRIORITY_MAX)
            .setCategory(NotificationCompat.CATEGORY_ALARM)
            .setOngoing(true)
            .setFullScreenIntent(pi, true)
            .setContentIntent(pi)
            .setVibrate(longArrayOf(0, 900, 600))
        val n = builder.build()
        n.flags = n.flags or Notification.FLAG_INSISTENT
        ctx.getSystemService(NotificationManager::class.java).notify(notifId(key), n)
    }
}

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import androidx.core.app.NotificationCompat

class ReminderReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        val key = intent.getStringExtra("key") ?: return
        Notifier.ensureChannels(ctx)
        val n = NotificationCompat.Builder(ctx, Notifier.CH_REMIND)
            .setSmallIcon(android.R.drawable.ic_popup_reminder)
            .setContentTitle(intent.getStringExtra("title") ?: "Напоминание")
            .setContentText("Пора")
            .setAutoCancel(true)
            .build()
        ctx.getSystemService(NotificationManager::class.java)
            .notify(Notifier.notifId(key), n)
    }
}
