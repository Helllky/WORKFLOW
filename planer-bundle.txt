# Планер дня — Android

Обёртка над веб-планером. Внутри WebView открывается страница из app/src/main/assets/index.html.
Будильники и напоминания ставятся системой через мост AndroidPlaner.

Сборка: при каждом коммите GitHub Actions собирает APK и публикует его в Releases.

const hsl = (h,s,l) => { s/=100; l/=100; const a = s*Math.min(l,1-l); return [0,8,4].map(n => { const k=(n+h/30)%12; return Math.round(255*(l - a*Math.max(-1,Math.min(k-3,9-k,1)))); }); };
function applyAccent(k){
  const a = ACC.find(x => x.k === (k || S.accent)) || ACC[0];
  const light = S.theme === 'light' || (S.theme === 'auto' && mq && mq.matches);
  const sat = a.s > 30 ? 1 : .25;
  const lum = c => { const f = v => { v /= 255; return v <= .03928 ? v/12.92 : Math.pow((v+.055)/1.055, 2.4); }; return .2126*f(c[0]) + .7152*f(c[1]) + .0722*f(c[2]); };
  let L = light ? 46 : 62, rgb = hsl(a.h, a.s * (light ? .85 : 1), L);
  // в светлой теме акцент темнеет, пока не станет читаемым на латте
  if (light) { const bg = lum(hsl(32,46,89)); while (L > 20 && (bg+.05)/(lum(rgb)+.05) < 3.2) { L -= 2; rgb = hsl(a.h, a.s*.85, L); } }
  const mx = (p,q,t) => p.map((v,i) => Math.round(v + (q[i]-v)*t));
  const tint = mx(hsl(a.h, Math.min(a.s,75)*.8, 58), [150,110,80], .3), kk = a.s > 30 ? 1 : .4;
  const bgL = [[252,246,238],[243,230,213],[232,212,189]].map((c,i) => `rgb(${mx(c, tint, (.10+.06*i)*kk)})`);
  const w = light ? '92,64,45' : '255,255,255'; // слои в светлой теме кофейные, не чёрные
  const V = light ? {
    '--bg1':bgL[0], '--bg2':bgL[1], '--bg3':bgL[2],
    '--o2':`rgb(${mx(rgb,[205,150,100],.45)})`, '--text':'#3b2a20', '--muted':'rgba(59,42,32,.62)',
    '--glass':`rgba(${mx([255,251,246],tint,.08)},.64)`, '--hl':'rgba(255,255,255,.55)', '--navbg':'rgba(255,248,240,.78)', '--sheet':'rgba(255,250,244,.97)',
    '--cs':'light', '--sh':'.16', '--orb':'.35', '--shc':'70,44,24'
  } : {
    '--bg1':`hsl(${a.h} ${50*sat}% 5%)`, '--bg2':`hsl(${a.h} ${42*sat}% 9%)`, '--bg3':`hsl(${a.h} ${40*sat}% 13%)`,
    '--o2':`hsl(${(a.h+40)%360} ${a.s}% 50%)`, '--text':'#f1f1f7', '--muted':'rgba(241,241,247,.55)',
    '--glass':'rgba(255,255,255,.05)', '--hl':'rgba(255,255,255,.06)', '--navbg':'rgba(30,30,55,.55)', '--sheet':'rgba(30,30,55,.9)',
    '--cs':'dark', '--sh':'.4', '--orb':'.5', '--shc':'0,0,0'
  };
  Object.assign(V, {'--accent':`rgb(${rgb})`, '--accent-rgb':rgb.join(','), '--w':w, '--border':`rgba(${w},${light?.16:.1})`});
  for (const n in V) document.documentElement.style.setProperty(n, V[n]);
}

function show(tab, forceDir){
  const prev = curTab();
  if (tab === prev && document.querySelector('#s-'+tab).classList.contains('active')) return;
  const dir = forceDir || (TABS.indexOf(tab) >= TABS.indexOf(prev) ? 'from-right' : 'from-left');
  document.querySelectorAll('.screen').forEach(s => {
    s.classList.remove('from-left','from-right');
    s.classList.toggle('active', s.id === 's-'+tab);
  });
  const el = document.querySelector('#s-'+tab);
  el.classList.add(dir);
  document.querySelectorAll('nav button').forEach(b => b.classList.toggle('on', b.dataset.tab === tab));
  document.querySelector('main').scrollTop = 0;
  currentTab = tab; if (tab !== 'settings') prevTab = tab;
  document.body.dataset.tab = tab;
  RENDER[tab] && RENDER[tab]();
}

const FALL_MS = 30*60000;
function addSleep(start, end, auto){
  const minutes = (end-start)/60000;
  if (minutes < 1) return;
  S.sleeps.push({id:uid(), date:dayKey(new Date(end)), start, end, minutes, auto});
  save();
}

function renderSleep(){
  const s = S.sleepNow;
  let html;
  if (!s){
    html = `<div class="h">Трекер сна</div>
      <div class="big" style="font-size:30px">Не спишь</div>
      <div class="muted" style="margin:6px 0 14px">Нажми, когда ложишься</div>
      <button class="btn" data-act="fall">Засыпаю</button>`;
  } else if (s.phase === 'falling'){
    const left = Math.max(0, FALL_MS - (Date.now()-s.fallStart));
    html = `<div class="h">Засыпаю</div>
      <div class="big" style="font-size:30px">${Math.ceil(left/60000)} мин</div>
      <div class="muted" style="margin:6px 0 14px">Отсчёт на засыпание, потом пойдёт сон</div>
      <button class="btn ghost" data-act="cancel">Передумал</button>`;
  } else {
    const el = (Date.now()-s.sleepStart)/60000;
    html = `<div class="h">Сплю</div>
      <div class="big" style="font-size:30px">${fmtDur(el)}</div>
      <div class="muted" style="margin:6px 0 14px">С ${hhmm(new Date(s.sleepStart).getHours()*60+new Date(s.sleepStart).getMinutes())}</div>
      <button class="btn" data-act="wake">Проснулся</button>`;
  }
  $('#sTrack').innerHTML = html; $('#homeSleep').innerHTML = html;

  const last7 = S.sleeps.slice(-7);
  const avg = last7.length ? last7.reduce((a,x) => a+x.minutes, 0)/last7.length : 0;
  let bars = '';
  for (let i = 6; i >= 0; i--){
    const d = new Date(Date.now() - i*864e5);
    const mins = sleepFor(dayKey(d));
    bars += `<div><i style="height:${Math.min(100, mins/540*100)}%;${mins?'':'opacity:.15'}"></i>${d.toLocaleDateString('ru-RU',{weekday:'short'})}</div>`;
  }
  $('#sStats').innerHTML = `
    <div class="muted">Средняя за последние ночи: <span style="color:var(--text)">${last7.length ? fmtDur(avg) : '—'}</span></div>
    <div class="week">${bars}</div>`;
}

/* ===================== СОН: ТАЙМЕР И АВТОРЕЖИМ ===================== */
function tickSleep(){
  const s = S.sleepNow;
  if (s && s.phase === 'falling' && Date.now() - s.fallStart >= FALL_MS){
    s.phase = 'asleep'; s.sleepStart = s.fallStart + FALL_MS; save();
  }
}

    if (away >= 60*60000 && !S.sleepNow) addSleep(S.hiddenAt + 60*60000, Date.now(), true);
    S.hiddenAt = null; save();
    renderSleep();
  }
});

const FREEZES=2, DAY_SHIFT=(5*60+21)*60000; // день для серии обновляется в 5:21, заморозок 2 в месяц
function streakDay(d){return dayKey(new Date((d||new Date()).getTime()-DAY_SHIFT));}
function streak(){
  const A=S.active||{},ks=Object.keys(A).sort(),sd=streakDay(),used={};
  const nx=(k,n)=>{const t=new Date(k+'T00:00:00');t.setDate(t.getDate()+n);return dayKey(t);};
  const cover=(f,t)=>{const need={};for(let k=f;k<=t;k=nx(k,1)){const m=k.slice(0,7);need[m]=(need[m]||0)+1;}
    for(const m in need)if((used[m]||0)+need[m]>FREEZES)return false;
    for(const m in need)used[m]=(used[m]||0)+need[m];return true;};
  let best=0,run=0,prev=null;
  ks.forEach(k=>{
    if(prev===null)run=1;
    else{const f=nx(prev,1),t=nx(k,-1);run=(f>t||cover(f,t))?run+1:1;}
    best=Math.max(best,run);prev=k;
  });
  let cur=0;
  if(ks.length){const last=ks[ks.length-1];
    if(last===sd)cur=run;
    else{const f=nx(last,1),t=nx(sd,-1);if(f>t||cover(f,t))cur=run;}}
  return {cur,best,today:!!A[sd],left:Math.max(0,FREEZES-(used[sd.slice(0,7)]||0))};
}
